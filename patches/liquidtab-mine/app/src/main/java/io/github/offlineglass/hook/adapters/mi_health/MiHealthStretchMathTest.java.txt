import io.github.offlineglass.hook.adapters.mi_health.MiHealthStretchMath;
public class MiHealthStretchMathTest {
 public static void main(String[] args) {
  for(float height:new float[]{600,2337,2560})for(float vector:new float[]{-.001f,-.1f,-.8f})for(float y:new float[]{-318,-120,0}) {
   // Android's negative-vector branch for inPos <= 0 samples y - distanceDiff.
   float nativeSample=y-(height/(1+Math.abs(vector))-height);
   float ourSample=y-MiHealthStretchMath.bottomTranslation(height,vector);
   if(Math.abs(nativeSample-ourSample)>.001f)throw new AssertionError("stretch sample mismatch");
  }
  if(MiHealthStretchMath.bottomTranslation(2337,0)!=0 || MiHealthStretchMath.bottomTranslation(2337,Float.NaN)!=0)throw new AssertionError();
  System.out.println("PASS: native inverse sampling above viewport, stretch release and invalid input");
 }
}
